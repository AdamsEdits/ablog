Frequent Tags

    {%- assign tag_counts = "" | split: "," -%} {%- for post in site.posts -%} {%- for tag in post.tags -%} {%- assign tag_counts = tag_counts | push: tag -%} {%- endfor -%} {%- endfor -%} {%- assign tags = site.posts | map: "tags" | join: "," | split: "," | uniq | sort -%} {%- for tag in tags -%} {%- if tag.size > 0 -%} {%- assign count = tag_counts | where_exp: "item", "item == tag" | size -%} {%- if count >= 2 -%}
  • {{ tag }} ({{ count }})
    • {%- endif -%} {%- endif -%} {%- endfor -%}
